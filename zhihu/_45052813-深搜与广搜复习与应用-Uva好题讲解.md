[Ideal Path - UVA 1599 - Virtual
Judge](https://link.zhihu.com/?target=https%3A//vjudge.net/problem/UVA-1599)

中文简化版翻译：

给出不定数量个输入

每个输入：

输入 n(节点个数),m（关系数量）

m个关系

x y z(无向图) z为边上的颜色号

输出

要求是1-n路径最短，并且是颜色号字典序最小的路径颜色序列

  

问题分析：

首先要找到最小路

其次最小路的边的颜色号字典序最小

  

该图可以看为边权为1的无向图，我们怎么去找到最短路径呢？

我们可以用宽搜。

有的同学就会想，我每次贪心加宽搜不就好了，可是局部最优不一定是全局最优，因为你选择的这条路不一定联通

那有的童鞋又说了，我深搜，直到找到目的然后，不断更换就好了，可以，但是会tle

那么我们应当怎么做呢？

首先我们看看深搜和宽搜的缺点：

宽搜不知道是否能够到达终点。不敢去贪心

深搜搜索范围太大，会浪费太多的时间

那么我们的题解是这样的：

从终点出发，宽搜，不断打上时间戳，相当于分层

从起点出发，按照时间戳访问回去，找到每层的最小值，有多个最小值要都考虑

  

解释：

分层是为什么 ： 找到最短路径，保证回去的时候是按照题目要求的路径走的

为什么终点出发，起点出发不行么？： 假设我们访问到了一条死胡同里，这条路是不是我们想要的呢？ 终点出发是为了一定能访问到起点

实现代码：

    
    
    #include<bits/stdc++.h>
    #define maxn 100010
    using namespace std;
    long long n,m;
    struct node
    {
    	long long to;
    	long long value;
    	node(long long y,long long z)
    	{
    		to = y;
    		value = z;
    	}
    };
    struct qu
    {
    	long long v;
    	long long c;
    	long long p;
    	qu(long long x,long long y,long long col)
    	{
    		v = x;
    		p = y;
    		c = col;
    	}
    };
    long long book[maxn];
    long long ans[maxn];
    bool ge[maxn];
    vector<node> v[maxn];
    void sbfs()
    {
    	memset(book,0,sizeof(book));
    	queue<long long> q;
    	book[n] = 1;
    	q.push(n);
    	while(!q.empty())
    	{
    		long long t = q.front();
    		q.pop();
    		long long si = v[t].size();
    		for(long long i=0; i<si; i++)
    		{
    			if(!book[v[t][i].to])
    			{
    				book[v[t][i].to] = book[t]+1;
    				q.push(v[t][i].to);
    			}
    		}
    	}
    }
    void  ebfs()
    {
    	queue<qu> q;
    	q.push(qu(1,0,0));
    	memset(ge,0,sizeof(ge));
    	while(!q.empty())
    	{
    		qu t = q.front();
    		q.pop();
    		if(t.c>ans[t.p]||ge[t.v])
    			continue;
    		ge[t.v] = true;
    		long long si = v[t.v].size();
    		for(long long i=0; i<si; i++)
    		{
    			if(book[v[t.v][i].to]==book[t.v]-1&&v[t.v][i].value<=ans[t.p+1])
    				q.push(qu(v[t.v][i].to,t.p+1,ans[t.p+1] = v[t.v][i].value));
    		}
    
    	}
    }
    int main()
    {
    	while(cin>>n>>m)
    	{
    		for(int i=1; i<=n; i++)
    			v[i].clear();
    		memset(ans,0x3f,sizeof(ans));
    		long long x,y,z;
    		for(long long i=0; i<m; i++)
    		{
    			cin>>x>>y>>z;
    			v[x].push_back(node(y,z));
    			v[y].push_back(node(x,z));
    		}
    		sbfs();
    		ebfs();
    		cout<<book[1]-book[n]<<endl;
    		for(long long i=1; i<book[1]-book[n]; i++)
    			cout<<ans[i]<<" ";
    		cout<<ans[book[1]-book[n]]<<endl;
    	}
    	return 0;
    }
