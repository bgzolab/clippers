一段典型的代码如下：

    
    
    import static com.whitewoodcity.xtrikeresources.clancy.Clancy.ACTIONS;
    import static com.whitewoodcity.xtrikeresources.clancy.Clancy.ORDERED_COMPONENTS;
    
    //把矢量图的数据装载进来
      private static final double FACTOR = .3;//这个缩放数据，1表示原始大小，矢量图不怕缩放
      public static BiMap<Component, ImageView> imageViewBiMap = HashBiMap.create();
      public static BiMap<Component, Rotates> rotatesBiMap = HashBiMap.create();
    
      static {
        for (var c : ORDERED_COMPONENTS) {
          var view = new JVG(c.getJsonString()).zoom(FACTOR).toImageView();//矢量图生成后转成像素图，方便处理，否则每一帧都要画一遍，机器压力比较大，容易卡
          imageViewBiMap.put(c, view);
          rotatesBiMap.put(c, new Rotates(view, c.rotateNum()));
        }
        for(var a: ACTIONS){
          var s = a.getJsonString();
          var json = new JsonObject(s);
          for(var c:ORDERED_COMPONENTS){
            switch (json.getValue(c.getFileName())){
              case JsonObject object ->  rotatesBiMap.get(c).record(a.getFileName(), object.toString(), FACTOR, true);
              case JsonArray array -> rotatesBiMap.get(c).record(a.getFileName(), array.toString(), FACTOR, true);
              default -> throw new RuntimeException(s+" is not a valid Json String");
            }
          }
        }
      }
